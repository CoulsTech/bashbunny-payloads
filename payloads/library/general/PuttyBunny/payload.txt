#!/bin/bash
#
# Title:         PuttyBunny
# Author:        CoulsTech
# Version:       0.1
# Target:        Windows 10/11
# Category:      General

# Setup
LED SETUP
ATTACKMODE HID

# Configuration
BAUD="115200"
DATA_BITS="8"
PARITY_BIT="No"
STOP_BIT="1"
FLOW_CONTROL="N"
BUNNY_USER="root"
BUNNY_PASS="hak5bunny"
COM_SLEEP="5"
# (to do) Install putty if not installed

# Payload
RUN WIN "powershell -NoExit"
QUACK "$PRESERIALBUNNY = [System.IO.Ports.SerialPort]::GetPortNames(); Start-Sleep -Seconds 1; $POSTSERIALBUNNY = [System.IO.Ports.SerialPort]::GetPortNames(); $BUNNYCOM = (Compare-Object $PRESERIALBUNNY $POSTSERIALBUNNY | Where-Object { $_.SideIndicator -eq '=>' }).InputObject"
QUACK ENTER
ATTACKMODE HID SERIAL #5 Second race for the A-B math (needs better solution)
QUACK DELAY "$COM_SLEEP*1000" (ms to second conversion)
QUACK "putty.exe -serial $BUNNYCOM -sercfg $BAUD,$DATA_BITS,$PARITY_BIT,$STOP_BIT,$FLOW_CONTROL"
QUACK ENTER
LED FINISH
