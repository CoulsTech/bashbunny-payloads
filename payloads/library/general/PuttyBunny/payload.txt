#!/bin/bash
#
# Title:         PuttyBunny
# Author:        CoulsTech
# Version:       0.1
# Target:        Windows 10/11
# Category:      General

# Setup
LED SETUP
ATTACKMODE HID SERIAL

# Configuration
COM_PORT="NULL"
BAUD="115200"
DATA_BITS="8"
PARITY_BIT="No"
STOP_BIT="1"
FLOW_CONTROL="N"
BUNNY_USER="root"
BUNNY_PASS="hak5bunny"

# (to do) Install putty if not installed

RUN WIN "powershell -WindowStyle Hidden -Command [System.IO.Ports.SerialPort::getportnames()
# (to do) define COM_PORT to catch the COM port from the victim machine

QUACK DELAY 10
if [$COM_PORT="NULL"]; then
    LED FAIL
else
    RUN WIN "putty.exe -serial $COM_PORT -sercfg $BAUD,$DATA_BITS,$PARITY_BIT,$STOP_BIT,$FLOW_CONTROL"
    LED FINISH
fi
