#!/bin/bash
#
# Title: BLE_CTL UART GATT Demo
# Author: CoulsTech
# Description: Demonstrates the E104-BT52 built-in UART BLE service through BLE_CTL.
# Target: Bash Bunny Mark II
# Attackmodes: SERIAL

LED SETUP
ATTACKMODE SERIAL

# The E104-BT52 does not expose a general-purpose BlueZ/GATT server.
# This configures its built-in UART service and two characteristics.
BLE_CTL GATT SERVER FFF0 FFF1 FFF2
sleep 2

BLE_CTL NAME "BB-BLE-UART"
BLE_CTL ADVERTISE on
BLE_CTL ADVERTISING_DATA "BB-BLE-UART"

LED B SLOW

# Pair/connect from nRF Connect or a serial BLE client, then write to the
# module's write characteristic. Incoming BLE payloads appear on /dev/ttyS1.
BLE_CTL RECEIVE 60 > /tmp/ble_ctl_demo_rx.txt

LED FINISH
